\section{Punto de partida}

\begin{frame}{Punto de partida}
\begin{block}{Por qué empezar por el lenguaje}
\begin{itemize}
  \item La matemática no empieza al calcular, sino al fijar significado.
  \item Una afirmación matemática necesita un contexto de interpretación.
  \item La notación ayuda solo cuando sus símbolos ya tienen significado.
\end{itemize}
\end{block}
\end{frame}

\begin{frame}{Punto de partida}
\begin{block}{Una frase ambigua}
\begin{center}
  \Large ``Un número grande tiene un cuadrado grande''
\end{center}
\vspace{0.2cm}
\begin{itemize}
  \item ¿Qué significa ``número''?
  \item ¿Respecto de qué orden se interpreta ``grande''?
  \item ¿En qué conjunto se está trabajando?
  \item ¿Qué operación se llama ``cuadrado''?
\end{itemize}
\end{block}
\end{frame}

\begin{frame}{Punto de partida}
\begin{block}{Una formulación con contexto}
Sea \(n\) un número natural. Si \(n>10\), entonces \(n^2>100\).
\end{block}

\begin{itemize}
  \item \(n\) representa un número natural.
  \item \(>\) es el orden usual en \(\N\).
  \item \(n^2\) significa \(n\cdot n\).
  \item La frase ya puede discutirse como enunciado matemático.
\end{itemize}
\end{frame}

\section{Lenguaje matemático}

\begin{frame}{Lenguaje matemático}
\begin{block}{Lenguaje natural}
\begin{itemize}
  \item Flexible, contextual y expresivo.
  \item Puede ser ambiguo si no se fijan los términos.
\end{itemize}
\end{block}

\begin{block}{Lenguaje matemático}
\begin{itemize}
  \item Combina palabras, símbolos y reglas de interpretación.
  \item Reduce ambigüedad mediante definiciones y convenciones.
\end{itemize}
\end{block}
\end{frame}

\begin{frame}{Lenguaje matemático}
\begin{block}{Principio de definición previa}
\begin{center}
  \Large Todo símbolo relevante debe definirse antes de utilizarse.
\end{center}
\end{block}

\begin{itemize}
  \item No es una regla decorativa: es una condición de lectura.
  \item Permite reconstruir qué significa cada paso.
  \item Obliga a declarar objetos, conjuntos, relaciones y operaciones.
\end{itemize}
\end{frame}

\begin{frame}{Lenguaje matemático}
\begin{block}{Definiciones, axiomas y teoremas}
\begin{itemize}
  \item Una definición introduce significado.
  \item Un axioma fija una propiedad aceptada como punto de partida.
  \item Un teorema afirma algo que debe demostrarse.
\end{itemize}
\end{block}

\begin{alertblock}{Idea de trabajo}
El orden de escritura importa: primero se fija el lenguaje, después se razona con él.
\end{alertblock}
\end{frame}

\section{Objetos y símbolos}

\begin{frame}{Objetos y símbolos}
\begin{block}{Objeto matemático}
Un objeto matemático es una entidad definida dentro de un contexto: número,
conjunto, función, relación, operación o estructura.
\end{block}

\begin{block}{Símbolo}
Un símbolo es un signo que representa un objeto, una relación, una operación o una propiedad.
\end{block}
\end{frame}

\begin{frame}{Objetos y símbolos}
\begin{block}{Anatomía de una expresión}
\[
  x\in A
\]
\begin{center}
\begin{tabular}{ccl}
\toprule
Símbolo & Tipo esperado & Papel \\
\midrule
\(x\) & objeto & posible elemento \\
\(A\) & conjunto & contenedor de objetos \\
\(\in\) & relación & pertenencia \\
\bottomrule
\end{tabular}
\end{center}
\end{block}
\end{frame}

\begin{frame}{Objetos y símbolos}
\begin{block}{Introducción correcta}
Sea \(A\) un conjunto y sea \(x\) un objeto. Escribiremos \(x\in A\) para indicar
que \(x\) pertenece al conjunto \(A\).
\end{block}

\begin{block}{Uso prematuro}
\[
  x\in A \Rightarrow x\in B
\]
La intención puede intuirse, pero faltan \(x\), \(A\), \(B\), \(\in\) y \(\Rightarrow\).
\end{block}
\end{frame}

\begin{frame}{Objetos y símbolos}
\begin{block}{Variables y constantes}
\begin{itemize}
  \item Una variable puede representar distintos objetos de un contexto fijado.
  \item Una constante representa un objeto fijo.
  \item El mismo símbolo puede cambiar de significado al cambiar de problema.
\end{itemize}
\end{block}

\begin{exampleblock}{Ejemplo}
En ``sea \(n\in\N\)'', \(n\) es variable. Una vez definido, \(0\) es una constante.
\end{exampleblock}
\end{frame}

\section{Lectura activa}

\begin{frame}{Lectura activa}
\begin{block}{Método de lectura de una fórmula}
\begin{enumerate}
  \item Localizar todos los símbolos.
  \item Decidir qué tipo de objeto representa cada uno.
  \item Identificar relaciones y operaciones.
  \item Reconstruir la frase en lenguaje natural.
  \item Comprobar si falta alguna definición previa.
\end{enumerate}
\end{block}
\end{frame}

\begin{frame}{Lectura activa}
\begin{block}{Ejemplo guiado}
\[
  A\subseteq B
\]
\begin{itemize}
  \item \(A\) y \(B\) deben ser conjuntos.
  \item \(\subseteq\) expresa inclusión, no pertenencia.
  \item Lectura: todo elemento de \(A\) es también elemento de \(B\).
\end{itemize}
\end{block}

\begin{alertblock}{Pregunta}
¿Qué habría que haber definido antes de escribir esta expresión?
\end{alertblock}
\end{frame}

\section{Trabajo de aula}

\begin{frame}{Trabajo de aula}
\begin{block}{Actividad 1: detectar símbolos no definidos}
Texto inicial:
\[
  x\in A \Rightarrow f(x)\in B.
\]
\begin{enumerate}
  \item Lista todos los símbolos.
  \item Indica qué tipo de objeto debería ser cada uno.
  \item Reescribe el texto introduciendo primero el contexto.
\end{enumerate}
\end{block}
\end{frame}

\begin{frame}{Trabajo de aula}
\begin{block}{Una posible reescritura}
Sean \(A\) y \(B\) dos conjuntos, sea \(f:A\to B\) una aplicación y sea \(x\in A\).
Entonces \(f(x)\in B\).
\end{block}

\begin{itemize}
  \item El dominio y el codominio de \(f\) quedan declarados.
  \item El símbolo \(x\) aparece después de fijar su contexto.
  \item La conclusión usa la información ya introducida.
\end{itemize}
\end{frame}

\begin{frame}{Cierre}
\begin{itemize}
  \item Una fórmula no es rigurosa si sus símbolos no han sido definidos.
  \item El contexto determina el significado de objetos, relaciones y operaciones.
  \item Leer matemáticas exige reconstruir significado, no solo reconocer símbolos.
  \item En la próxima clase: definiciones, enunciados, ejemplos y demostraciones.
\end{itemize}
\end{frame}

\begin{frame}{Trabajo recomendado}
\begin{itemize}
  \item Leer las secciones 1.1 a 1.4 del manual.
  \item Subrayar cada símbolo nuevo y anotar cuándo se define.
  \item Reescribir dos frases ambiguas con contexto matemático explícito.
\end{itemize}
\end{frame}
